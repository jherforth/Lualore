[lualore]

# Enable automatic cave castle placement in newly generated underground areas.
lualore_cave_castles (Cave castles) bool true

# Grid spacing (in nodes) between cave castle candidates. One candidate exists
# per spacing x spacing area, jittered in the middle half of the cell, so two
# castles can never be closer together than half this value.
lualore_cave_castle_spacing (Cave castle spacing) int 400 200 5000

# Fraction of grid cells that actually attempt to spawn a castle (0.0 - 1.0).
lualore_cave_castle_chance (Cave castle spawn chance) float 1.0 0.0 1.0

# Cave castles are only placed on cave floors at or below this Y level.
lualore_cave_castle_y_top (Cave castle spawn band top) int -120 -31000 0

# Cave castles are only placed on cave floors at or above this Y level
# (lower number = deeper).
lualore_cave_castle_y_bottom (Cave castle spawn band bottom) int -1200 -31000 0

# Carve the castle crypt (statue room + stairwell) open after placement so the
# wizard boss group can always spawn. Turn off only for debugging.
lualore_cave_castle_carve (Cave castle carve crypt) bool true
lualore_cave_village (Cave castle abandoned village) bool true
lualore_cave_village_radius (Cave castle village radius) int 46 26 80
lualore_cave_village_houses (Cave castle village houses) int 9 3 16
lualore_cave_village_chests (Cave castle village chests) bool true

# -------------------------------------------------------------------
# Villages (deterministic grid placement - see villagers/VILLAGE_PLACEMENT.md)
# -------------------------------------------------------------------

# Enable automatic village placement in newly generated areas.
lualore_villages (Villages) bool true

# Grid spacing (in nodes) between village candidates. One candidate exists
# per spacing x spacing area, jittered in the middle half of the cell, so two
# villages can never be closer together than half this value.
# Lower = more villages (400 = roughly one village per 400x400 area on average).
lualore_village_spacing (Village spacing) int 320 200 5000

# Fraction of grid cells that actually attempt to spawn a village (0.0 - 1.0).
lualore_village_chance (Village spawn chance) float 0.9 0.0 1.0
lualore_village_terraform (Village terraforming) bool true
lualore_village_terraform_max (Village terraform band) int 8 2 16

# Give the terraformed site a lobed, irregular outline instead of a circle.
lualore_village_organic (Village organic outline) bool true

# Scatter patches of accent ground blocks (worn dirt, gravel, dry grass ...)
# over the levelled village floor instead of leaving one uniform surface.
lualore_village_ground_noise (Village ground texture) bool true

# Grass, flowers and shrubs around the buildings.
lualore_village_vegetation (Village vegetation) bool true

# How thickly the vegetation grows (0.0 = none, 1.0 = every free block).
# Each biome scales this by its own factor - deserts stay sparse.
lualore_village_plant_density (Village plant density) float 0.22 0.0 1.0

# Trodden paths from every building to the village centre.
lualore_village_paths (Village paths) bool true

# Minimum number of houses per village (placed on a ring around the centre).
lualore_village_houses_min (Village houses minimum) int 5 1 20

# Maximum number of houses per village (placed on a ring around the centre).
lualore_village_houses_max (Village houses maximum) int 20 1 20

# How far houses may be placed from the village centre, in nodes.
lualore_village_radius (Village house radius) int 22 8 60

# Chance for each central building (church / market / stable) per village
# (0.0 - 1.0). Space limits may still leave some out.
lualore_village_central_chance (Village central building chance) float 0.45 0.0 1.0

# Villages are searched for between these Y levels (surface band).
lualore_village_y_max (Village search band top) int 200 -31000 31000

# Lower Y limit of the village search band (lower number = deeper).
lualore_village_y_min (Village search band bottom) int -2 -31000 31000

# -------------------------------------------------------------------
# Ruins (procedural broken structures - see ruins/RUINS.md)
# -------------------------------------------------------------------

# Enable automatic ruin placement in newly generated areas.
lualore_ruins (Ruins) bool true

# Grid spacing (in nodes) between ruin candidates. One candidate exists per
# spacing x spacing area, jittered in the middle half of the cell, so two
# ruins can never be closer together than half this value.
# Lower = more ruins (450 = roughly one ruin per 450x450 area on average).
lualore_ruin_spacing (Ruin spacing) int 450 200 5000

# Fraction of grid cells that actually attempt to spawn a ruin (0.0 - 1.0).
lualore_ruin_chance (Ruin spawn chance) float 0.75 0.0 1.0

# Let some ruins contain treasure chests (filled with biome loot by the
# existing loot system).
lualore_ruin_chests (Ruin treasure chests) bool true

# Ruins are searched for between these Y levels (surface band).
lualore_ruin_y_max (Ruin search band top) int 200 -31000 31000

# Lower Y limit of the ruin search band (lower number = deeper).
lualore_ruin_y_min (Ruin search band bottom) int -2 -31000 31000

# -------------------------------------------------------------------
# Sky portals (see portal/SKY_PORTAL.md)
# -------------------------------------------------------------------

# Mese block frames that light when an iron ingot is dropped into them,
# leading to the sky islands and back. Portals already in a world stay
# visible when this is off; they simply stop lighting and carrying anyone.
lualore_sky_portals (Sky portals) bool true

# Seconds a player has to stand in a portal before it carries them.
lualore_portal_delay (Sky portal delay) float 2.0 0.0 10.0

# Height from which a portal counts as being in the sky: portals at or
# above it lead down, portals below it lead up, and the search for an
# island starts here. Asuna's floatlands begin at y = 287.
lualore_portal_sky_y (Sky portal sky height) int 290 0 30000

# A portal on the ground leads to a sky village. If none a portal founded
# lies within this many nodes, it raises a crystal island overhead and
# founds one; otherwise it goes to the nearest. Lower it for more villages.
lualore_portal_village_reach (Sky portal village reach) int 800 100 10000

# Houses in a village a portal founds. The sky-site grid builds hamlets of
# 2-6; a portal village is a town. The island grows to fit.
lualore_portal_village_houses (Sky portal village houses) int 18 6 30

# Sky Folk a portal town is filled to whenever a player arrives through a
# portal (at most once every 30 seconds per town). Until its valkyries are
# defeated a town's Sky Folk are hostile, so a high number makes an
# unfreed town dangerous.
lualore_portal_village_folk (Sky portal town population) int 40 0 120

# Largest interior width or height the frame scan will accept. Frames have
# no real size limit; this only stops the scan following an endless wall of
# mese.
lualore_portal_max_size (Sky portal maximum size) int 256 3 1024

# Mirror Sentinels wander the sky islands (above lualore_portal_sky_y).
lualore_mirror_sentinels (Mirror Sentinels) bool true

# -------------------------------------------------------------------
# Sky sites (deterministic grid placement on the floating crystal
# islands - see valkyrie/SKY_SITES.md)
# -------------------------------------------------------------------

# Enable deterministic placement of the sky buildings on the floating
# crystal islands (replaces the old schematic decorations, which stacked
# castles and houses on top of each other).
lualore_sky_buildings (Sky buildings) bool true

# Grid spacing (in nodes) between sky site candidates. One candidate exists
# per spacing x spacing area, jittered in the middle half of the cell, so
# two sky sites can never be closer together than half this value. Cells
# whose candidate finds no island simply build nothing.
lualore_sky_spacing (Sky site spacing) int 360 160 5000

# Fraction of grid cells that actually attempt a sky site (0.0 - 1.0).
lualore_sky_chance (Sky site chance) float 0.8 0.0 1.0

# Chance that a sky site is built around a skycastle. The castle holds the
# one valkyrie chest, so this is what makes it a rare find.
lualore_sky_castle_chance (Skycastle chance) float 0.15 0.0 1.0

# Houses per sky site (the castle, when one is rolled, is extra).
lualore_sky_houses_min (Sky houses minimum) int 2 1 12
lualore_sky_houses_max (Sky houses maximum) int 6 1 12

# How far from the site centre houses may stand, in nodes. Grows with the
# rolled house count and again when the site has a castle.
lualore_sky_radius (Sky site radius) int 30 14 60

# Scatter crystal grass and lay paths between the sky buildings.
lualore_sky_dressing (Sky site dressing) bool true

# Sky sites are searched for between these Y levels.
lualore_sky_y_min (Sky search band bottom) int 500 -31000 31000
lualore_sky_y_max (Sky search band top) int 31000 -31000 31000

# -------------------------------------------------------------------
# Villager jobs (see villagers/VILLAGER_JOBS.md)
# -------------------------------------------------------------------

# Villagers claim a workstation near their bed and spend their working
# day at it - the blacksmith at his anvil, the fisherman at his trap -
# returning home at dusk as before. Turning this off restores the old
# wander/socialise day exactly.
lualore_villager_jobs (Villager jobs) bool true

# Place the workstations no building schematic contains - the anvil and
# forge, the cleric's altar, and one or two tilled fields - into each new
# village. Use /furnish_village to add them to a village that already
# exists.
lualore_village_workstations (Village workstations) bool true

# Per-player standing with each village. Feeding, trading and using what
# villagers make earns it, and better standing means a larger share of
# what they produce. Earn-only: nothing you do can lower it.
lualore_village_standing (Village standing) bool true

# -------------------------------------------------------------------
# Village commissions (see villagers/VILLAGE_EXPANSION.md)
# -------------------------------------------------------------------

# Players commission new houses at a village's notice board, deliver the
# materials, and a newcomer of the trade they chose moves in.
lualore_village_commissions (Village commissions) bool true

# Most houses one village can be given by commission.
lualore_village_max_commissions (Commissioned houses per village) int 8 0 30

# How much more each commission costs than the last (0.25 = 25% more).
lualore_commission_cost (Commission cost growth) float 0.25 0.0 2.0
